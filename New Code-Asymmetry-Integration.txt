check_asym_map <- function(A, land.pairs, midline = integer(0)) {
  p <- dim(A)[1]
  if (!is.matrix(land.pairs) || ncol(land.pairs) != 2)
    stop("land.pairs must be a two-column integer matrix (left,right).")

  if (nrow(land.pairs) < 1)
    stop("land.pairs must have at least one row.")

  if (any(!is.finite(land.pairs))) stop("land.pairs contains non-finite values.")

  if (any(land.pairs < 1 | land.pairs > p))
    stop("Indices in land.pairs out of range 1..", p)

  if (length(midline)) {
    if (any(!is.finite(midline))) stop("midline contains non-finite values.")
    if (any(midline < 1 | midline > p))
      stop("Indices in midline out of range 1..", p)
  }

  # no overlap between midline and any paired landmark
  if (length(midline) && length(intersect(midline, as.vector(land.pairs))) > 0)
    stop("Some midline landmarks also appear in land.pairs; please separate them.")

  invisible(TRUE)
}

# Validators for 2BPLS block definitions
.check_one_2bpls <- function(A, blocks, name = "2BPLS") {
  stopifnot(is.list(blocks), length(blocks) == 2)
  p <- dim(A)[1]

  for (nm in names(blocks)) {
    idx <- blocks[[nm]]
    if (!is.integer(idx)) idx <- as.integer(idx)
    if (length(idx) < 2) stop("[", name, "] Block '", nm, "' must have ≥ 2 landmarks.")
    if (any(is.na(idx))) stop("[", name, "] Block '", nm, "' has NA indices.")
    if (any(idx < 1 | idx > p)) stop("[", name, "] Block '", nm, "' indices out of range 1..", p)
    blocks[[nm]] <- unique(idx)
  }

  # disjointness within the comparison
  if (length(intersect(blocks[[1]], blocks[[2]])) > 0) {
    ov <- intersect(blocks[[1]], blocks[[2]])
    stop("[", name, "] Blocks share landmarks (must be disjoint). Overlap: ",
         paste(sort(ov), collapse = ", "))
  }
  invisible(TRUE)
}

# ----- helper to pull scores from a two.b.pls object safely -----
.get_pls_scores <- function(pls) {
  # Try the common geomorph names first
  Xcands <- c("XScores", "XScoresZ", "XScores1", "X.scores")
  Ycands <- c("YScores", "YScoresZ", "YScores1", "Y.scores")
  pick_first <- function(obj, cands){
    for (nm in cands) if (!is.null(obj[[nm]])) return(as.matrix(obj[[nm]]))
    NULL
  }
  X <- pick_first(pls, Xcands)
  Y <- pick_first(pls, Ycands)
  if (is.null(X) || is.null(Y))
    stop("Could not find X/ Y scores in the PLS object. Inspect names(pls).")
  list(X = X, Y = Y)
}

# ----- PLS axis-1 plot colored by Sex, with optional hulls/ellipses -----
plot_pls1_by_sex <- function(pls, meta_df, ids_in_pls,
                             out_path, sex_cols,
                             add_hulls = TRUE, add_ellipses = TRUE) {
  sc <- .get_pls_scores(pls)
  # first axis scores
  if (ncol(sc$X) < 1 || ncol(sc$Y) < 1) stop("PLS scores have no columns.")
  x1 <- sc$X[, 1, drop = TRUE]
  y1 <- sc$Y[, 1, drop = TRUE]

  # build plotting frame aligned to the order used in PLS
  df <- data.frame(ID = ids_in_pls, PLS1_X = as.numeric(x1), PLS1_Y = as.numeric(y1),
                   stringsAsFactors = FALSE)
  # align meta by ID (assumes meta_df has an ID column)
  if (!"ID" %in% names(meta_df)) stop("meta_df must contain column 'ID'.")
  meta_al <- meta_df[match(df$ID, meta_df$ID), , drop = FALSE]
  df$Sex <- droplevels(factor(meta_al$Sex))
  # keep only M/F (you can tweak if you also want 'X')
  keep <- !is.na(df$Sex) & df$Sex %in% c("M","F")
  df <- df[keep, , drop = FALSE]
  df$Sex <- droplevels(df$Sex)
  if (!nrow(df)) stop("No rows with Sex == M/F for PLS plot.")

  # plot
  plt <- ggplot2::ggplot(df, ggplot2::aes(PLS1_X, PLS1_Y, color = Sex)) +
    ggplot2::geom_point(pch = 19, size = 2.8, alpha = 0.95) +
    ggplot2::theme_light() +
    ggplot2::xlab("PLS axis 1 (X block)") +
    ggplot2::ylab("PLS axis 1 (Y block)") +
    ggplot2::scale_color_manual(values = sex_cols, drop = FALSE)

  # optional convex hulls by Sex
  if (add_hulls) {
    hulls <- try({
      make_hulls(df, "Sex", x = "PLS1_X", y = "PLS1_Y",
                 exclude_levels = c("X"))
    }, silent = TRUE)
    if (!inherits(hulls, "try-error") && nrow(hulls)) {
      plt <- plt + ggplot2::geom_polygon(
        data = hulls,
        ggplot2::aes(PLS1_X, PLS1_Y, group = Sex, fill = Sex),
        alpha = 0.12, inherit.aes = FALSE, linewidth = 0.4, color = NA
      ) +
      ggplot2::scale_fill_manual(values = sex_cols, drop = FALSE)
    }
  }

  # optional 95% ellipses per Sex
  if (add_ellipses && nlevels(df$Sex) >= 2) {
    plt <- plt + ggplot2::stat_ellipse(level = 0.95, linewidth = 0.6, alpha = 0.7)
  }

  ggplot2::ggsave(out_path, plt, dpi = 600, width = 10, height = 5)
  invisible(out_path)
}

# Single-sheet Excel (robust to empty data)
write_xlsx1 <- function(df, path) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  wb <- openxlsx::createWorkbook()
  openxlsx::addWorksheet(wb, "Sheet1")
  if (is.data.frame(df) && ncol(df) == 0) {
    df <- data.frame(note = "empty table (no columns)", check.names = FALSE)
  }
  openxlsx::writeData(wb, "Sheet1", df)
  openxlsx::saveWorkbook(wb, path, overwrite = TRUE)
}

# Robust TXT writer
write_txt1 <- function(lines, path) {
  dir.create(dirname(path), recursive = TRUE, showWarnings = FALSE)
  con <- file(path, open = "wt", encoding = "UTF-8")
  on.exit(try(close(con), silent = TRUE), add = TRUE)
  writeLines(lines, con)
}

ensure_dirs <- function(...) {
  dirs <- list(...)
  for (d in dirs) if (!dir.exists(d)) dir.create(d, recursive = TRUE, showWarnings = FALSE)
  invisible(NULL)
}

# Error wrapper
safe_call <- function(expr, label = "step", outdir = NULL, stop_on_error = TRUE) {
  tryCatch(
    force(expr),
    error = function(e) {
      msg <- paste0("[", label, "] FAILED: ", conditionMessage(e))
      if (!is.null(outdir)) {
        dir.create(outdir, recursive = TRUE, showWarnings = FALSE)
        writeLines(msg, file.path(outdir, "errors.log"), useBytes = TRUE)
      }
      if (stop_on_error) stop(msg) else message(msg)
      invisible(NULL)
    }
  )
}

# ---------- Species/subspecies mapping (self-contained) ----------
.normalize_key <- function(x){
  x <- tolower(trimws(as.character(x)))
  if (x %in% c("", "na", "nan")) return(NA_character_)
  x
}
map_label_to_species <- function(x){
  x <- .normalize_key(x)
  if (is.na(x)) return(NA_character_)
  if (grepl("^cam", x))  return("cam")       # camelopardalis
  if (grepl("^ret", x))  return("ret")       # reticulata
  if (grepl("^tip", x))  return("tip")       # tippelskirchi
  if (grepl("^gir", x))  return("gir")       # giraffa
  if (grepl("^sp",  x))  return("sp.")       # unspecified
  # collapse subspecies to parent species
  if (grepl("^ant|^per|^cam\\.", x)) return("cam")
  if (grepl("^ang|^gir\\.", x))      return("gir")
  if (grepl("^thor", x))             return("tip")
  NA_character_
}
species_from_any <- function(Species, Subspecies){
  s  <- map_label_to_species(Species); if (!is.na(s)) return(s)
  ss <- map_label_to_species(Subspecies); if (!is.na(ss)) return(ss)
  "sp."
}
make_species_plot_factor <- function(meta_df){
  sp <- vapply(seq_len(nrow(meta_df)), function(i){
    species_from_any(meta_df$Species[i], meta_df$Subspecies[i])
  }, FUN.VALUE = character(1))
  factor(sp, levels = c("cam","ret","tip","gir","sp."))
}

}

# LOWER-triangle pairwise p matrix, with optional level exclusions
pairwise_p_matrix <- function(X, group, iter = 9999, exclude_levels = NULL) {
  X <- as.data.frame(X)
  stopifnot(nrow(X) == length(group))

  # drop rows with NA group or NA in any column
  keep <- !is.na(group)
  if (ncol(X) > 0) keep <- keep & rowSums(!is.na(X)) == ncol(X)
  X <- X[keep, , drop = FALSE]
  group <- droplevels(group[keep])

  # optionally exclude certain levels (e.g., "sp.")
  if (!is.null(exclude_levels)) {
    keep2 <- !(group %in% exclude_levels)
    X <- X[keep2, , drop = FALSE]
    group <- droplevels(group[keep2])
  }

  lev <- levels(group); k <- length(lev)
  if (k < 2) return(data.frame(group = lev, check.names = FALSE))

  M <- matrix(NA_real_, k, k, dimnames = list(lev, lev))
  for (i in 1:(k-1)) for (j in (i+1):k) {
    sel <- group %in% lev[c(i, j)]
    G2  <- droplevels(group[sel])
    Y2  <- as.matrix(X[sel, , drop = FALSE])
    DF  <- geomorph.data.frame(Y = Y2, G = G2)
    fit <- procD.lm(Y ~ G, data = DF, iter = iter, print.progress = FALSE)
    # term is "G" here (in this small DF), because the formula uses G
    M[j, i] <- suppressWarnings(summary(fit)$table["G", "Pr(>F)"])
  }
  M[upper.tri(M, diag = TRUE)] <- NA_real_
  cbind(group = rownames(M), as.data.frame(M, check.names = FALSE))
}

# Save + directly color p < threshold (no conditionalFormatting)
write_pairwise_excel <- function(
  X, group, path, iter = 9999,
  p_threshold = 0.05,
  fill_hex = "#C6E0B4",
  exclude_levels = NULL
){
  mat_df <- pairwise_p_matrix(X, group, iter = iter, exclude_levels = exclude_levels)

  # ensure numeric columns are numeric
  is_num <- vapply(mat_df, is.numeric, logical(1))
  for (j in which(is_num)) mat_df[[j]] <- as.numeric(mat_df[[j]])

  wb <- openxlsx::createWorkbook()
  sh <- "Sheet1"
  openxlsx::addWorksheet(wb, sh)
  openxlsx::writeData(wb, sh, mat_df)

  # number format for numeric cells
  num_style <- openxlsx::createStyle(numFmt = "0.0000")
  data_rows <- 2:(nrow(mat_df) + 1)
  num_cols  <- which(is_num)
  if (length(num_cols)) {
    openxlsx::addStyle(
      wb, sh, style = num_style,
      rows = data_rows, cols = num_cols,
      gridExpand = TRUE, stack = TRUE
    )
  }

  # paint significant cells
  sig_style <- openxlsx::createStyle(fgFill = fill_hex)
  if (length(num_cols)) {
    for (ri in seq_len(nrow(mat_df))) {
      for (cj in num_cols) {
        val <- mat_df[[cj]][ri]
        if (is.finite(val) && val < p_threshold) {
          openxlsx::addStyle(
            wb, sh, style = sig_style,
            rows = ri + 1, cols = cj, stack = TRUE
          )
        }
      }
    }
  }

  openxlsx::saveWorkbook(wb, path, overwrite = TRUE)
}

# Return a p-value from a procD.lm / lm.rrpp summary table, robust to naming
get_pr_p <- function(tab, term = c("G","Group")) {
  if (inherits(tab, "try-error") || is.null(tab)) return(NA_real_)
  rn <- rownames(tab)
  term <- intersect(term, rn)
  if (!length(term)) return(NA_real_)
  # columns vary by version
  pcols <- c("Pr(>F)", "Pr(>F)1", "Pr(>F)2", "P-value", "P.value", "P")
  pcol  <- intersect(pcols, colnames(tab))
  if (!length(pcol)) return(NA_real_)
  as.numeric(tab[term[1], pcol[1]])
}

# For pairwise tests only: drop placeholder/unknown levels

  droplevels(G[!G %in% bad])
}

# Compute DA/FA (handles geomorph version differences)
compute_asymmetry <- function(coords, land.pairs, midline = integer(0), iter = 9999){
  ind <- factor(dimnames(coords)[[3]])
  fml_names <- names(formals(geomorph::bilat.symmetry))
  args <- list(A = coords, ind = ind, land.pairs = land.pairs,
               iter = iter, print.progress = FALSE)
  if ("object.sym" %in% fml_names)           args$object.sym <- TRUE
  else if ("object.symmetry" %in% fml_names) args$object.symmetry <- TRUE
  if (length(midline) > 0 && "midline" %in% fml_names) args$midline <- midline

  fit <- do.call(geomorph::bilat.symmetry, args)

  get_comp <- function(fit, candidates){
    for (nm in candidates) if (!is.null(fit[[nm]])) return(fit[[nm]])
    NULL
  }
  sym  <- get_comp(fit, c("shape.S","symm.shape","S","Sym","sym"))
  asym <- get_comp(fit, c("shape.A","asymm.shape","A","Asym","asym"))

  smry <- try(suppressWarnings(summary(fit)), silent = TRUE)
  anova_tab <- if (!inherits(smry, "try-error") && !is.null(smry$ANOVA)) smry$ANOVA else NA
  Rsq_val   <- if (!inherits(smry, "try-error") && !is.null(smry$Rsq))   smry$Rsq   else NA

  list(
    fit       = fit,
    fit_names = names(fit),
    anova     = anova_tab,
    R2        = Rsq_val,
    sym       = sym,
    asym      = asym
  )
}

# Coerce 2D to p×k×n array using template coords dims
coerce_to_shape_array <- function(X, template_coords, label = "shape") {
  if (length(dim(X)) == 3) return(X)
  p <- dim(template_coords)[1]; k <- dim(template_coords)[2]; n <- dim(template_coords)[3]
  if (is.null(p) || is.null(k) || is.null(n))
    stop(label, ": cannot infer (p,k,n) from template coords.")
  if (is.vector(X)) stop(label, ": got a vector; cannot reshape to array.")
  if (!is.matrix(X))
    stop(label, ": expected matrix or array; got class ", paste(class(X), collapse = ","))

  if (nrow(X) == n && ncol(X) == p * k) return(geomorph::arrayspecs(X, p, k))
  if (nrow(X) == p * k && ncol(X) == n) return(geomorph::arrayspecs(t(X), p, k))
  stop(label, ": cannot coerce to p×k×n. Got ", paste(dim(X), collapse = "×"),
       " but expected ", n, "×", p*k, " or ", p*k, "×", n, ".")
}

# DA magnitude per group
DA_magnitude_by_group <- function(shapeA, group){
  group <- droplevels(group)
  lev <- levels(group)
  mags <- sapply(lev, function(lv){
    A <- shapeA[, , group == lv, drop = FALSE]
    DA <- mshape(A)
    sqrt(sum(DA^2))
  })
  data.frame(Group = lev, DA_mag = mags, row.names = NULL, check.names = FALSE)
}

# FA magnitudes per specimen
FA_magnitudes <- function(shapeA, group){
  group <- droplevels(group)
  lev <- levels(group)
  mags <- numeric(length(group))
  for (lv in lev){
    idx <- which(group == lv)
    A   <- shapeA[, , idx, drop = FALSE]
    DA  <- mshape(A)
    mags[idx] <- apply(A, 3, function(M) sqrt(sum((M - DA)^2)))
  }
  mags
}

run_asymmetry_block <- function(coords, meta_df, outdir, sample_name,
                                land.pairs, midline = integer(0), iter = 9999){

  dir.create(outdir, recursive = TRUE, showWarnings = FALSE)
  if (missing(land.pairs) || is.null(land.pairs) || nrow(as.matrix(land.pairs)) < 1) {
    write_txt1("Skipped: no land.pairs provided.", file.path(outdir, "Asymmetry_debug.txt"))
    return(invisible(TRUE))
  }

  ids <- dimnames(coords)[[3]]
  if (is.null(ids)) stop("coords must have specimen dimnames")

  # Choose & align grouping; drop NAs
  G0 <- safe_choose_group(sample_name, meta_df)
  if (!is.null(rownames(meta_df))) G0 <- G0[match(ids, rownames(meta_df))]

  # Enforce N/S order only if chosen group actually equals Region
  if ("Region" %in% names(meta_df) && !is.null(rownames(meta_df))) {
    R <- droplevels(meta_df$Region[match(ids, rownames(meta_df))])
    if (isTRUE(all.equal(G0, R))) G0 <- factor(G0, levels = c("North","South"))
  }
  G0 <- droplevels(G0)

  keep <- !is.na(G0)
  coords2 <- coords[, , keep, drop = FALSE]
  G       <- droplevels(G0[keep])

  if (length(G) < 2 || nlevels(G) < 2 || dim(coords2)[3] < 2) {
    write_txt1(c(
      paste0("Symmetry — ", sample_name),
      sprintf("After filtering: n=%d, levels=%s", dim(coords2)[3], paste(levels(G), collapse=", ")),
      "Not enough data to proceed."
    ), file.path(outdir, "Asymmetry_debug.txt"))
    return(invisible(TRUE))
  }

  # Compute asymmetry
  asy <- compute_asymmetry(coords2, land.pairs = land.pairs, midline = midline, iter = iter)

  if (is.null(asy$asym)) {
    write_txt1(c(
      "Asymmetry error: asymmetric component not found.",
      "Names in fit:", paste(asy$fit_names, collapse = ", ")
    ), file.path(outdir, "Asymmetry_debug.txt"))
    return(invisible(TRUE))
  }

  asy$asym <- try(coerce_to_shape_array(asy$asym, coords2, "shape.A"), silent = TRUE)
  if (inherits(asy$asym, "try-error")) {
    write_txt1(c("Coercion error for shape.A:", as.character(asy$asym)),
               file.path(outdir, "Asymmetry_debug.txt"))
    return(invisible(TRUE))
  }
  if (!is.null(asy$sym)) {
    asy$sym <- try(coerce_to_shape_array(asy$sym, coords2, "shape.S"), silent = TRUE)
  }

  dims_msg <- sprintf("shape.A dims: p=%d, k=%d, n=%d",
                      dim(asy$asym)[1], dim(asy$asym)[2], dim(asy$asym)[3])

  # Summary
  s_obj <- try(summary(asy$fit), silent = TRUE)
  txt <- capture.output({
    cat("Symmetry analysis — ", sample_name, "\n\n", sep = "")
    cat(dims_msg, "\n\n", sep = "")
    if (!inherits(s_obj, "try-error")) print(s_obj) else
      cat("NOTE: summary() failed; writing ANOVA and downstream stats only.\n")
  })
  write_txt1(txt, file.path(outdir, "Asymmetry_summary.txt"))

  # ANOVA table
  an_df <- try(as.data.frame(asy$anova), silent = TRUE)
  if (inherits(an_df, "try-error") || !nrow(an_df) || !ncol(an_df)) {
    write_txt1("ANOVA table was empty.", file.path(outdir, "Asymmetry_ProcrustesANOVA.txt"))
  } else {
    an_df$Effect <- rownames(an_df)
    ord <- intersect(c("Effect","Df","SS","MS","Rsq","F","Z","Pr(>F)","Pr(>F)1","Pr(>F)2"),
                     colnames(an_df))
    if (!length(ord)) ord <- colnames(an_df)
    write_xlsx1(an_df[, ord, drop = FALSE],
                file.path(outdir, "Asymmetry_ProcrustesANOVA.xlsx"))
  }

  write_txt1(c(
    dims_msg, "Group summary:", paste(capture.output(print(summary(G))), collapse = "\n")
  ), file.path(outdir, "Asymmetry_debug.txt"))

  # ---- DA: procD.lm on asymmetric shapes + pairwise
  da_try <- try({
    gdfD <- geomorph.data.frame(Y = asy$asym, Group = G)
    fitD <- procD.lm(Y ~ Group, data = gdfD, iter = iter, print.progress = FALSE)
    tabD <- as.data.frame(summary(fitD)$table); tabD$Effect <- rownames(tabD)
    write_xlsx1(tabD[, c("Effect", setdiff(names(tabD), "Effect")), drop = FALSE],
                file.path(outdir, "DA_between_groups_ANOVA.xlsx"))

    # ----- DA pairwise (lower triangle), drop placeholders only for pairwise -----
    G_pair_levels <- levels(drop_placeholder_levels(G))  # just the levels to keep
    lev <- G_pair_levels; k <- length(lev)
    M <- matrix(NA_real_, k, k, dimnames = list(lev, lev))

    if (k >= 2) {
      for (i in 1:(k-1)) for (j in (i+1):k) {
        # specimens that belong to either level i or j, using ORIGINAL G
        idx <- which(G %in% lev[c(i, j)])
        if (length(idx) < 4) next  # need data
        g2 <- droplevels(G[idx])
        if (nlevels(g2) < 2 || min(table(g2)) < 2) next

        Y2 <- asy$asym[, , idx, drop = FALSE]
        if (length(idx) != dim(Y2)[3]) next  # hard guard

        d2  <- geomorph.data.frame(Y = Y2, G = g2)
        fit <- procD.lm(Y ~ G, data = d2, iter = iter, print.progress = FALSE)
        tab <- summary(fit)$table
        M[j, i] <- suppressWarnings(get_pr_p(tab, term = c("G","Group")))
      }
    }
    M[upper.tri(M, diag = TRUE)] <- NA_real_
    write_pairwise_matrix_with_highlight(
  cbind(group = rownames(M),  as.data.frame(M,  check.names = FALSE)),
  file.path(outdir, "DA_pairwise_matrix.xlsx"),
  p_threshold = 0.05)
  }, silent = TRUE)

  if (inherits(da_try, "try-error")) {
    write_txt1(c("DA model failed.", as.character(da_try)),
               file.path(outdir, "DA_errors.txt"))
  }

  # ---- FA: per-specimen magnitudes + tests (robust)
  IDs2   <- dimnames(asy$asym)[[3]]
  FA_mag <- FA_magnitudes(asy$asym, G)
  dfF <- data.frame(
    ID    = IDs2,
    FA    = as.numeric(FA_mag),
    Group = droplevels(factor(G)),
    stringsAsFactors = FALSE
  )

  # keep only finite FA + non-missing groups
  keepF <- is.finite(dfF$FA) & !is.na(dfF$Group)
  dfF   <- droplevels(dfF[keepF, , drop = FALSE])

  # keep groups with at least 2 specimens
  gcounts <- table(dfF$Group)
  good_levels <- names(gcounts[gcounts >= 2])
  dfF <- droplevels(dfF[dfF$Group %in% good_levels, , drop = FALSE])

  # Export per-specimen FA magnitudes (filtered)
  write_xlsx1(dfF[, c("ID","Group","FA")],
              file.path(outdir, "FA_magnitudes_per_specimen.xlsx"))

  if (nlevels(dfF$Group) < 2) {
    write_txt1(c(
      "FA model skipped: need ≥2 groups with ≥2 specimens each.",
      paste("Group counts (after filtering):",
            paste(names(gcounts), as.integer(gcounts), sep="=", collapse=", "))
    ), file.path(outdir, "FA_errors.txt"))
  } else {
    # Global FA test
    fa_try <- try({
      fitF <- RRPP::lm.rrpp(FA ~ Group, data = dfF, iter = iter, print.progress = FALSE)
      tabF <- as.data.frame(summary(fitF)$table); tabF$Effect <- rownames(tabF)
      write_xlsx1(tabF[, c("Effect", setdiff(names(tabF), "Effect")), drop = FALSE],
                  file.path(outdir, "FA_between_groups_ANOVA.xlsx"))
    }, silent = TRUE)

    if (inherits(fa_try, "try-error")) {
      write_txt1(c(
        "FA model failed (lm.rrpp).",
        "Underlying error:",
        as.character(fa_try)
      ), file.path(outdir, "FA_errors.txt"))
    } else {
      # ----- FA pairwise on filtered dfF (lower triangle), dropping placeholders -----
      gpw_levels <- levels(drop_placeholder_levels(dfF$Group))
      dfF2 <- droplevels(dfF[dfF$Group %in% gpw_levels, , drop = FALSE])
      levF <- levels(dfF2$Group); kF <- length(levF)
      MF <- matrix(NA_real_, kF, kF, dimnames = list(levF, levF))

      if (kF >= 2) {
        for (i in 1:(kF-1)) for (j in (i+1):kF) {
          sel <- which(dfF2$Group %in% levF[c(i, j)])
          if (length(sel) < 4) next
          G2 <- droplevels(dfF2$Group[sel])
          if (nlevels(G2) < 2 || min(table(G2)) < 2) next

          dsub <- dfF2[sel, , drop = FALSE]
          fit  <- RRPP::lm.rrpp(FA ~ Group, data = dsub, iter = iter, print.progress = FALSE)
          tab  <- summary(fit)$table
          MF[j, i] <- suppressWarnings(get_pr_p(tab, term = "Group"))
        }
      }
      MF[upper.tri(MF, diag = TRUE)] <- NA_real_
      write_pairwise_matrix_with_highlight(
  cbind(group = rownames(MF), as.data.frame(MF, check.names = FALSE)),
  file.path(outdir, "FA_pairwise_matrix.xlsx"),
  p_threshold = 0.05)
    }
  }

  invisible(TRUE)
}

# Extract a landmark block
extract_block <- function(coords, lm_idx){
  coords[lm_idx, , , drop = FALSE]
}

# Partition factor for modularity.test
.make_partition_from_blocks <- function(coords, blocks){
  p <- dim(coords)[1]
  part <- rep(NA_character_, p)
  stopifnot(length(blocks) == 2)
  nm <- names(blocks); if (is.null(nm)) nm <- c("Block1","Block2")
  b1 <- unique(as.integer(blocks[[1]]))
  b2 <- unique(as.integer(blocks[[2]]))
  if (length(intersect(b1, b2))) stop("Blocks for modularity share landmarks.")
  if (any(b1 < 1 | b1 > p) || any(b2 < 1 | b2 > p)) stop("Block indices out of range.")
  part[b1] <- nm[1]; part[b2] <- nm[2]
  part[is.na(part)] <- "Unassigned"
  factor(part)
}

# Modularity runner
run_modularity_for <- function(coords, blocks, outdir, name, iter = 9999){
  dir.create(outdir, recursive = TRUE, showWarnings = FALSE)
  partition <- .make_partition_from_blocks(coords, blocks)
  modres <- geomorph::modularity.test(coords, partition.gp = partition,
                                      iter = iter, print.progress = FALSE)
  write_txt1(capture.output({cat("Modularity —", name, "\n\n"); print(modres)}),
             file.path(outdir, paste0("Modularity_", name, "_summary.txt")))

  rv <- z <- pval <- NA_real_
  sm <- try(suppressWarnings(summary(modres)), silent = TRUE)
  if (!inherits(sm, "try-error")) {
    if (!is.null(sm$RV)) rv <- sm$RV
    if (!is.null(sm$Z))  z  <- sm$Z
    if (!is.null(sm$P.value)) pval <- sm$P.value
  }
  write_xlsx1(data.frame(Comparison=name, RV=rv, Z=z, `Pr(>RV)`=pval, iter=iter,
                         check.names=FALSE),
              file.path(outdir, paste0("Modularity_", name, "_summary.xlsx")))
  invisible(TRUE)
}

# --------- WARPER for PLS1 ends (saves OBJ) ----------
export_pls_warps_for_block <- function(
  A_raw, coords_gpa, blocks, pls_obj, block_id, out_dir,
  ref_id, ref_mesh_obj, warp_mag_sd = 2
){
  p <- dim(coords_gpa)[1]; k <- dim(coords_gpa)[2]
  ids <- dimnames(coords_gpa)[[3]]
  i  <- which(ids == ref_id); if (!length(i)) return(invisible(FALSE))
  iR <- which(dimnames(A_raw)[[3]] == ref_id); if (!length(iR)) return(invisible(FALSE))
  ref_raw <- A_raw[,,iR, drop=FALSE][,,1]

  mesh <- rgl::readOBJ(ref_mesh_obj)  # mesh3d
  # pull PLS1 scores for the chosen block
  get_scores <- function(o, x=TRUE){
    for (nm in if (x) c("XScores","Xscores") else c("YScores","Yscores"))
      if (!is.null(o[[nm]])) return(as.numeric(o[[nm]][,1]))
    stop("PLS scores missing")
  }
  scores <- if (block_id == 1) get_scores(pls_obj, TRUE) else get_scores(pls_obj, FALSE)
  sds <- stats::sd(scores)

  block_idx <- blocks[[block_id]]
  coords_block <- coords_gpa[block_idx, , , drop = FALSE]
  Xb  <- geomorph::two.d.array(coords_block)
  fit <- lm(Xb ~ scores)
  beta <- coef(fit)[2, ]                 # (p_block*k) vector
  dB   <- geomorph::arrayspecs(beta, length(block_idx), k)

  Dfull <- matrix(0, nrow=p, ncol=k); Dfull[block_idx, ] <- dB
  delta <- Dfull * (warp_mag_sd * sds) * geomorph::centroid.size(ref_raw)

  nm <- sprintf("PLS1_Block%d", block_id)
  Rvcg::vcgExport(Morpho::tps3d(mesh, ref = ref_raw, target = ref_raw - delta),
                  filename = file.path(out_dir, paste0(nm, "_negative.obj")))
  Rvcg::vcgExport(Morpho::tps3d(mesh, ref = ref_raw, target = ref_raw + delta),
                  filename = file.path(out_dir, paste0(nm, "_positive.obj")))
  invisible(TRUE)
}

# 2BPLS runner (scores, pairwise, plot) + WARPS
run_2bpls_for <- function(coords, meta_df, blocks, outdir, name, sample_name,
                          A_raw, ref_id, ref_mesh_obj, iter = 9999){
  stopifnot(is.list(blocks), length(blocks) == 2)
  dir.create(outdir, recursive = TRUE, showWarnings = FALSE)

  Ablock <- extract_block(coords, blocks[[1]])
  Bblock <- extract_block(coords, blocks[[2]])

  pls <- geomorph::two.b.pls(Ablock, Bblock, iter = iter, print.progress = FALSE)
  s   <- summary(pls)
  write_txt1(capture.output(print(s)),
             file.path(outdir, paste0("2BPLS_", name, "_summary.txt")))

  get_scores <- function(obj, x = TRUE){
    for (nm in if (x) c("XScores","Xscores") else c("YScores","Yscores"))
      if (!is.null(obj[[nm]])) return(obj[[nm]])
    NULL
  }
  xs <- get_scores(pls, TRUE);  ys <- get_scores(pls, FALSE)
  if (is.null(xs) || is.null(ys) || nrow(xs) == 0 || nrow(ys) == 0)
    stop("PLS returned empty scores.")

  ids <- rownames(xs); if (is.null(ids)) ids <- dimnames(coords)[[3]]
  sc <- data.frame(
    ID           = ids,
    PLS1_Block1  = as.numeric(xs[, 1, drop = TRUE]),
    PLS1_Block2  = as.numeric(ys[, 1, drop = TRUE]),
    check.names  = FALSE, stringsAsFactors = FALSE
  )
  rownames(sc) <- sc$ID

    meta2 <- meta_df[match(sc$ID, rownames(meta_df)), , drop = FALSE]
  if (any(is.na(rownames(meta2)))) stop("Could not match PLS IDs to metadata.")

  # ----- FORCE grouping by Sex (M/F only) -----
  sx <- toupper(as.character(meta2$Sex))
  sx[!(sx %in% c("M","F"))] <- NA
  G <- factor(sx, levels = c("M","F"))
  sc$Group <- droplevels(G)

  # write scores with Sex grouping
  write_xlsx1(sc, file.path(outdir, paste0("2BPLS_", name, "_scores.xlsx")))

  write_txt1(c("Group levels for plotting (Sex only):",
               paste(levels(droplevels(sc$Group)), collapse = ", ")),
             file.path(outdir, paste0("2BPLS_", name, "_group_levels.txt")))

  # pairwise now (by Sex)
  write_pairwise_excel(
    sc[, c("PLS1_Block1","PLS1_Block2")],
    sc$Group,
    file.path(outdir, paste0("2BPLS_", name, "_pairwise_matrix.xlsx")),
    iter = iter
  )

  # Sex palette (uses your global sex_cols)
  lev <- levels(sc$Group)
  pal <- if (length(lev)) sex_cols[lev] else sex_cols
  ttl <- sprintf("2BPLS — %s — %s (r-PLS=%.3f; P=%.4f)", sample_name, name, s$r.pls, s$P.value)

  plot_xy_by_group(
    sc, "PLS1_Block1", "PLS1_Block2", "Group",
    pal, ttl,
    file.path(outdir, paste0("2BPLS_", name, "_PLS1_vs_PLS1.jpg"))
  )

  export_pls_warps_for_block <- function(
  A_raw, coords_gpa, blocks, pls_obj, block_id, out_dir,
  ref_id, ref_mesh_obj, warp_mag_sd = 2
){
  dir.create(out_dir, recursive = TRUE, showWarnings = FALSE)

  p <- dim(coords_gpa)[1]; k <- dim(coords_gpa)[2]
  ids <- dimnames(coords_gpa)[[3]]
  i   <- which(ids == ref_id); if (!length(i)) return(invisible(FALSE))
  iR  <- which(dimnames(A_raw)[[3]] == ref_id); if (!length(iR)) return(invisible(FALSE))

  ref_raw <- A_raw[,, iR, drop = FALSE][,,1]
  mesh    <- rgl::readOBJ(ref_mesh_obj)

  # pull PLS1 scores for the chosen block
  get_scores <- function(o, x = TRUE){
    cands <- if (x) c("XScores","Xscores","XScores1","XScoresZ") else c("YScores","Yscores","YScores1","YScoresZ")
    for (nm in cands) if (!is.null(o[[nm]])) return(as.numeric(o[[nm]][,1]))
    stop("PLS scores missing for block ", if (x) "X" else "Y")
  }
  scores <- if (block_id == 1) get_scores(pls_obj, TRUE) else get_scores(pls_obj, FALSE)
  sds    <- stats::sd(scores, na.rm = TRUE)
  if (!is.finite(sds) || sds == 0) sds <- 1

  block_idx    <- blocks[[block_id]]
  coords_block <- coords_gpa[block_idx, , , drop = FALSE]
  Xb           <- geomorph::two.d.array(coords_block)

  # linear mapping PLS1 -> shape change for that block (regression on scores)
  fit  <- stats::lm(Xb ~ scores)
  beta <- stats::coef(fit)[2, ]                  # (p_block * k) vector
  dB   <- geomorph::arrayspecs(beta, length(block_idx), k)

  # embed delta into full landmark matrix (only block moves)
  Dfull <- matrix(0, nrow = p, ncol = k)
  Dfull[block_idx, ] <- dB

  # scale displacement by warp_mag_sd * sd(PLS1) * centroid size (keeps magnitude sensible)
  cs_ref <- geomorph::centroid.size(ref_raw)
  delta  <- Dfull * (warp_mag_sd * sds) * cs_ref

  # targets in RAW
  tar_neg <- ref_raw - delta
  tar_pos <- ref_raw + delta

  nm_base <- sprintf("PLS1_Block%d", block_id)

  # ---- TPS warp meshes (as before) ----
  mN <- Morpho::tps3d(mesh, ref = ref_raw, target = tar_neg)
  mP <- Morpho::tps3d(mesh, ref = ref_raw, target = tar_pos)
  Rvcg::vcgExport(mN, filename = file.path(out_dir, paste0(nm_base, "_negative.obj")))
  Rvcg::vcgExport(mP, filename = file.path(out_dir, paste0(nm_base, "_positive.obj")))

  # ---- NEW: export RAW landmark coordinates for the two endpoints ----
  utils::write.table(tar_neg,
    file = file.path(out_dir, paste0(nm_base, "_negative_rawXYZ.csv")),
    row.names = FALSE, col.names = FALSE, sep = ","
  )
  utils::write.table(tar_pos,
    file = file.path(out_dir, paste0(nm_base, "_positive_rawXYZ.csv")),
    row.names = FALSE, col.names = FALSE, sep = ","
  )

  invisible(TRUE)
}


  invisible(TRUE)
}


# Wrapper: BOTH comparisons + modularity (+ WARPS)
run_2bpls_block <- function(A_raw, coords, meta_df, dataset_dir, sample_name, iter = 9999){
  mod_root <- file.path(dataset_dir, "Modularity")
  vd_dir   <- file.path(mod_root, "Ventral_vs_Dorsal")
  nf_dir   <- file.path(mod_root, "Neuro_vs_Face")
  ensure_dirs(mod_root, vd_dir, nf_dir)

  # --- drop bad IDs (if any) for this dataset ---
  ids_now <- dimnames(coords)[[3]]
  bad_now <- BAD_2BPLS_IDS[[sample_name]]
  if (length(bad_now)) {
    keep <- !(ids_now %in% bad_now)
    if (!all(keep)) {
      coords  <- coords[,, keep, drop = FALSE]
      # re-align metadata to the filtered coords order
      meta_df <- meta_df[match(dimnames(coords)[[3]], rownames(meta_df)), , drop = FALSE]
      # small audit note
      writeLines(
        c("2BPLS exclusions:", paste("Removed IDs:", paste(bad_now, collapse = ", "))),
        file.path(dataset_dir, "2BPLS_exclusions.txt")
      )
    }
  }

  # choose reference after filtering
  ref <- choose_ref_2bpls(sample_name, coords, A_raw)

  # Ventral vs Dorsal
  safe_call(
    run_2bpls_for(coords, meta_df, blocks_2BPLS_ventral_dorsal,
                  vd_dir, "Ventral_vs_Dorsal", sample_name,
                  A_raw = A_raw, ref_id = ref$ref_id, ref_mesh_obj = ref$ref_mesh, iter = iter),
    paste0("2BPLS Ventral vs Dorsal (", sample_name, ")"),
    vd_dir, stop_on_error = FALSE
  )
  safe_call(
    run_modularity_for(coords, blocks_2BPLS_ventral_dorsal,
                       vd_dir, "Ventral_vs_Dorsal", iter = iter),
    paste0("Modularity Ventral vs Dorsal (", sample_name, ")"),
    vd_dir, stop_on_error = FALSE
  )

  # Neuro vs Face
  safe_call(
    run_2bpls_for(coords, meta_df, blocks_2BPLS_neuro_face,
                  nf_dir, "Neuro_vs_Face", sample_name,
                  A_raw = A_raw, ref_id = ref$ref_id, ref_mesh_obj = ref$ref_mesh, iter = iter),
    paste0("2BPLS Neuro–Face (", sample_name, ")"),
    nf_dir, stop_on_error = FALSE
  )
  safe_call(
    run_modularity_for(coords, blocks_2BPLS_neuro_face,
                       nf_dir, "Neuro_vs_Face", iter = iter),
    paste0("Modularity Neuro–Face (", sample_name, ")"),
    nf_dir, stop_on_error = FALSE
  )
}

run_sample_asym2bpls <- function(A, meta, sample_name, iter = 9999){
  # Folder layout
  parent_dir   <- file.path(base_out, sample_name)
  complete_sub <- .complete_subfolder_name(sample_name)
  outdir       <- if (is.null(complete_sub)) parent_dir else file.path(parent_dir, complete_sub)

  asym_dir  <- file.path(outdir, "Asymmetry")
  integ_dir <- file.path(outdir, "Integration")  # base for 2BPLS/Modularity/WARPS

  ensure_dirs(parent_dir, outdir, asym_dir, integ_dir)
  cat("\n=== Asymmetry/Integration:", sample_name, "→", normalizePath(outdir, mustWork = FALSE), "===\n")

  cat("\n=== Asym/2BPLS Sample:", sample_name, "===\n")

  # Checks
  if (is.null(dim(A)) || length(dim(A)) != 3)
    stop("Array `A` must be a 3D landmark array for sample ", sample_name)
  needed_cols <- c("ID","Species","Subspecies","Sex","Age","Region")
  miss_cols <- setdiff(needed_cols, names(meta))
  if (length(miss_cols)) stop("Metadata missing columns: ", paste(miss_cols, collapse=", "))

  # GPA
  gp <- gpagen(A, curves = sliders, ProcD = FALSE, print.progress = TRUE, PrinAxes = FALSE)
  coords <- gp$coords
  ids_gp <- dimnames(coords)[[3]]

  # Align metadata
  if (!all(ids_gp %in% meta$ID)) {
    stop("Some GPA specimen IDs are missing from metadata for ", sample_name,
         "\nMissing: ", paste(ids_gp[!ids_gp %in% meta$ID], collapse = ", "))
  }
  meta2 <- meta[match(ids_gp, meta$ID), , drop = FALSE]
  if (any(meta2$ID != ids_gp)) stop("Metadata did not align 1:1 to GPA IDs for ", sample_name)
  meta_df <- as.data.frame(meta2); rownames(meta_df) <- meta_df$ID

  # Asymmetry block → .../Asymmetry/
  safe_call(
    run_asymmetry_block(coords, meta_df, asym_dir,
                        sample_name, land.pairs = land.pairs, midline = midline, iter = iter),
    paste0("Asymmetry (", sample_name, ")"),
    asym_dir, stop_on_error = FALSE
  )

  # 2BPLS + Modularity + WARPS → .../Integration/
  safe_call(
    run_2bpls_block(A_raw = A, coords = coords, meta_df = meta_df,
                    dataset_dir = integ_dir, sample_name = sample_name, iter = iter),
    paste0("2BPLS+Modularity (", sample_name, ")"),
    integ_dir, stop_on_error = FALSE
  )
}
